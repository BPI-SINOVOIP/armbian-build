"""只核對已完成的小型證據；不連線、不操作媒體或電源。"""
import hashlib
import json
from pathlib import Path
import re
import sys
import time

HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[3]
BASE=HERE.parent
sys.path.insert(0,str(ROOT/'tools'))
from bpi_h618_lab_summary import summarize
import bpi_h618_artifacts as safe
from bpi_h618_lab_progress import decoded


def read(path):
    with safe.open_root(path.parent) as parent:
        ref,blob=safe.fingerprint(parent,path.name,limit=2*1024**2,keep=True)
    return decoded(blob),{'path':str(path),**ref}


def main():
    result={'ok':False,'started_unix':time.time(),'media_written':False,
        'hardware_operated':False,'complete_boot_chain_per_cycle_sha256_verified':False,
        'independent_rescue_memory_hour_verified':False,'cold_returns':[]}
    snapshot=summarize(BASE)
    assert not snapshot['read_errors'] and snapshot['counts']['images']==10
    result['counts']=snapshot['counts']
    inputs,ref=read(HERE/'rescue-inputs-002.json')
    result['rescue_inputs']=ref
    roles=list(zip(('40080000','4fa00000','45000000','4ff00000'),
        (inputs['inputs'][i] for i in (0,1,2,4))))
    paths=[BASE/'T3-bookworm-return-001/report.json']
    paths+=sorted(BASE.glob('T4-matrix-*/*/return*/report.json'))
    nonces,timestamps,hashes=set(),set(),set()
    for path in paths:
        item,source=read(path)
        assert all(item.get(k) is True for k in ('ok','recovery_verified',
            'independent_root_ram_verified','strict_ssh_verified','sd_prefix_unchanged'))
        assert item['identity']['kernel']=='6.6.75-current-sunxi64'
        assert item['sd_prefix']==inputs['sd_prefix'] and item['off_seconds']>=10
        inventory,inventory_ref=read(path.parent/'ssh-inventory.json')
        assert inventory_ref['sha256']==item['inventory_sha256'] and inventory==item['inventory']
        assert len(inventory['swaps'].splitlines())==1
        assert '/dev/mmcblk' not in inventory['mountinfo']
        assert any(d.get('device/cid')=='d629034339413535311299942ee08c07'
            and d.get('bytes')==31289507840 for d in inventory['devices'])
        power=item['power']
        assert [p['command'] for p in power]==['status','off','status','on']
        for entry,on in zip(power,(True,False,False,True)):
            response=entry['result']; device=response['device']
            assert entry['returncode']==0 and response['ok'] is True and response['verified'] is True
            assert device['identity_verified'] is True and device['name']=='bpi-pw-1'
            assert device['mac']=='EC:B9:31:24:F7:D1' and device['ip']=='192.168.50.245'
            assert device['on'] is on
        stamp=power[-1]['result']['timestamp_unix_ms']
        assert stamp-power[1]['result']['timestamp_unix_ms']>=10000
        nonce=item['commands'][0]['supervisor']['nonce']
        assert item['commands'][0]['supervisor']['abi']==3
        assert item['commands'][1]['nonce_hex']==f'{nonce:08x}'
        assert nonce not in nonces and stamp not in timestamps and source['sha256'] not in hashes
        nonces.add(nonce);timestamps.add(stamp);hashes.add(source['sha256'])
        commands={c['command']:c for c in item['commands'] if 'command' in c}
        checks=[]
        for address,blob in roles:
            load=commands[f'load mmc 0:1 {address} {blob["path"]}']
            crc=commands[f'crc32 {address} {blob["bytes"]:x}']
            length=re.findall(r'(?m)^(\d+) bytes read in ',load['output'])
            actual=re.findall(r'(?m)^crc32 for [^\r\n]*==> ([0-9a-f]{8})\r?$',crc['output'])
            assert load['ok'] is True and crc['ok'] is True
            assert length==[str(blob['bytes'])] and actual==[blob['crc32']]
            checks.append({'path':blob['path'],'bytes':blob['bytes'],'crc32':actual[0]})
        assert 'booti 40080000 4ff00000:1ad94db 4fa00000' in commands
        hardware_id=None
        if path.relative_to(BASE).parts[0].startswith('T4-'):
            batch,_=read(BASE/path.relative_to(BASE).parts[0]/'summary.json')
            assert batch['board_serial']=='0845'
            hardware_id='0845'
        result['cold_returns'].append({'source':source,'hardware_id':hardware_id,
            'nonce':nonce,'on_timestamp_unix_ms':stamp,'off_seconds':item['off_seconds'],
            'normal_shutdown_verified':item.get('normal_shutdown_verified') is True,
            'forced_poweroff':item.get('forced_poweroff') is True,'loaded_checks':checks,
            'inventory_sha256':inventory_ref['sha256']})
    result.update(ok=True,finished_unix=time.time(),cold_return_count=len(paths),
        t4_explicit_board_serial_cold_return_count=sum(x['hardware_id']=='0845' for x in result['cold_returns']),
        source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest())
    out=BASE/'T6-first-pass-001'
    out.mkdir(mode=0o700,exist_ok=False)
    for name,data in [('summary.json',snapshot),('audit.json',result)]:
        with (out/name).open('x') as stream:
            json.dump(data,stream,ensure_ascii=False,indent=2)
    print(json.dumps({k:v for k,v in result.items() if k!='cold_returns'},ensure_ascii=False,indent=2))


if __name__=='__main__':
    main()
