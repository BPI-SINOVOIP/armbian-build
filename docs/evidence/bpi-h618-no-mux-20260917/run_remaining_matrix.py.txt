"""0845 剩餘九套基本測試；部署、開機、短測、回救援循序執行，不重做首套。"""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[3]
sys.path.insert(0,str(ROOT/'tools'))
import bpi_h618_emmc_deploy as deploy
import bpi_h618_customer_smoke as smoke

BASE=ROOT/'output/evidence/bpi-h618-no-mux'
OUT=BASE/'T4-matrix-005'
PY=sys.executable
INDEX=BASE/'T0-customer-components-001/components-index.json'


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def save(path,value):
    temp=path.with_suffix(path.suffix+'.tmp')
    with temp.open('x') as stream:
        json.dump(value,stream,ensure_ascii=False,indent=2)
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(temp,path)


def main():
    OUT.mkdir(mode=0o700,exist_ok=False)
    index=json.loads(INDEX.read_text())
    if index['count']!=10 or index['ok'] is not True: raise RuntimeError('十套索引未完成')
    first=json.loads((BASE/'T3-bookworm-return-001/report.json').read_text())
    first_receipt=json.loads((BASE/'T3-bookworm-deploy-001/receipt.json').read_text())
    if not first.get('recovery_verified') or first_receipt['status']!='verified':
        raise RuntimeError('首套沒有完整部署與回救援證據')
    summary={'schema':'bpi-h618-lab-batch-v1','board_serial':'0845','started_unix':time.time(),
             'status':'running','index_sha256':digest(INDEX),'cases':[],
             'reused':[{'os':'bookworm','desktop':'minimal','raw_sha256':first_receipt['source']['raw']['sha256'],
                       'return_report_sha256':digest(BASE/'T3-bookworm-return-001/report.json'),
                       'all_tests_passed':False}],
             'scope':'基本系統層；普通使用者、桌面登入、GPU、周邊與長測未納入本批次',
             'original_boot_chain_verified':False,'private_helpers':{}}
    for name in ('customer_cycle.py','customer_access.py','return_to_rescue.py','rescue_network.py','rescue_cycles.py'):
        summary['private_helpers'][name]=digest(HERE/name)
    rescue_config=first['ssh_config']
    xfce_return=BASE/'T4-matrix-003/bookworm-xfce_desktop/return/report.json'
    xfce=json.loads(xfce_return.read_text())
    if not xfce.get('recovery_verified') or not xfce.get('sd_prefix_unchanged'):
        raise RuntimeError('第二套回救援證據不完整')
    summary['reused'].append({'os':'bookworm','desktop':'xfce_desktop',
        'return_report_path':str(xfce_return),'return_report_sha256':digest(xfce_return),
        'all_tests_passed':False})
    for os_name,desktop in [('jammy','minimal'),('jammy','xfce_desktop'),('noble','minimal')]:
        path=BASE/'T4-matrix-004'/(os_name+'-'+desktop)/'return/report.json'
        completed=json.loads(path.read_text())
        if not completed.get('ok') or not completed.get('recovery_verified') or not completed.get('sd_prefix_unchanged'):
            raise RuntimeError('既有案例回救援證據不足')
        summary['reused'].append({'os':os_name,'desktop':desktop,'return_report_path':str(path),
            'return_report_sha256':digest(path),'all_tests_passed':False})
    save(OUT/'summary.json',summary)
    try:
        for entry in index['entries']:
            if entry['os'] in ('bookworm','jammy') or (entry['os'],entry['desktop'])==('noble','minimal'):
                continue
            key=entry['os']+'-'+entry['desktop']
            manifest=Path(entry['manifest']['path'])
            if digest(manifest)!=entry['manifest']['sha256']: raise RuntimeError('組件清單已變更')
            item=json.loads(manifest.read_text())
            case=OUT/key
            case.mkdir(mode=0o700)
            record={'key':key,'image':item['image'],'components_sha256':entry['manifest']['sha256'],
                    'status':'running','stage':'deploy','started_unix':time.time(),'events':[]}
            summary['cases'].append(record)
            def stage(name):
                record['stage']=name
                record['events'].append({'stage':name,'started_unix':time.time()})
                save(OUT/'summary.json',summary)
                print(json.dumps({'case':key,'stage':name},ensure_ascii=False),flush=True)
            def command(name,argv,timeout):
                for helper,expected in summary['private_helpers'].items():
                    if digest(HERE/helper)!=expected: raise RuntimeError('執行輔助程式版本變更，停止本批次')
                stage(name)
                with (case/(name+'-stdout.log')).open('xb') as stdout, (case/(name+'-stderr.log')).open('xb') as stderr:
                    result=subprocess.run(argv,cwd=ROOT,stdout=stdout,stderr=stderr,timeout=timeout)
                record['events'][-1].update(exitcode=result.returncode,finished_unix=time.time())
                save(OUT/'summary.json',summary)
                if result.returncode: raise RuntimeError(key+' 的 '+name+' 未成功，保留現場，不盲目重試')
            previous=BASE/'T4-matrix-004/noble-xfce_desktop'
            deploy_receipt=case/'deploy/receipt.json'
            if key=='noble-xfce_desktop':
                receipt=json.loads((previous/'deploy/receipt.json').read_text())
                recheck=json.loads((previous/'readback-after-hang-001/report.json').read_text())
                if receipt.get('ok') is not True or receipt.get('status')!='verified' or receipt['source']['raw']!=item['image']['raw'] or recheck.get('ok') is not True or recheck['remote']['raw']!=item['image']['raw']:
                    raise RuntimeError('異常後唯讀核對未通過，拒絕重試')
                deploy_receipt=previous/'deploy/receipt.json'
                record['reused_stages']={'deploy':{'path':str(deploy_receipt),'sha256':digest(deploy_receipt)}}
                record['readback_after_hang']={'path':str(previous/'readback-after-hang-001/report.json'),
                    'sha256':digest(previous/'readback-after-hang-001/report.json')}
                record.update(deploy_verified=True,cold_boot_retry=1)
                stage('resume_boot')
            else:
                stage('deploy')
                receipt=deploy.deploy(confirm_overwrite=True,
                backup_manifest=BASE/'T2-0845-emmc-backup-001/manifest.json',
                source=item['image']['path'],compressed_sha256=item['image']['compressed']['sha256'],
                raw_sha256=item['image']['raw']['sha256'],raw_size=item['image']['raw']['bytes'],
                expected_cid='d629034339413535311299942ee08c07',expected_size=31289507840,
                expected_controller='/sys/devices/platform/soc/4022000.mmc',
                protected_sd_cid='03534453523634478697bc8c0701846b',
                protected_sd_controller='/sys/devices/platform/soc/4020000.mmc',
                ssh_config=rescue_config,alias='bpi-rescue-0845',output_dir=case/'deploy',timeout=1800)
                record['deploy_verified']=receipt['ok'] and receipt['status']=='verified'
                record['events'][-1]['finished_unix']=time.time()
            try:
                command('boot',[PY,'-B',str(HERE/'customer_cycle.py'),'--components',str(manifest),
                    '--components-sha256',entry['manifest']['sha256'],'--receipt',str(deploy_receipt),
                    '--output',str(case/'boot'),'--initialize'],500)
            except RuntimeError as error:
                failed_path=case/'boot/report.json'
                failed=json.loads(failed_path.read_text())
                if failed.get('ok') is not False or not failed.get('error') or not failed.get('finished_unix'):
                    raise
                record['boot_error']=str(error)
                command('return',[PY,'-B',str(HERE/'return_to_rescue.py'),str(case/'return'),
                    '--failed-boot-report',str(failed_path),'--failed-boot-sha256',digest(failed_path)],360)
                recovery=json.loads((case/'return/report.json').read_text())
                if not recovery.get('recovery_verified') or not recovery.get('sd_prefix_unchanged'):
                    raise RuntimeError('失敗後回救援未核對，不進下一套')
                rescue_config=recovery['ssh_config']
                record.update(status='boot_failed_recovered',recovery_verified=True,
                    finished_unix=time.time(),all_hardware_tests_passed=False,desktop_login_tested=False)
                save(OUT/'summary.json',summary)
                continue
            boot=json.loads((case/'boot/report.json').read_text())
            record['root_initialized']=boot.get('root_initialized',False)
            command('access',[PY,'-B',str(HERE/'customer_access.py'),str(case/'access')],120)
            config=str(case/'access/ssh-config')
            stage('smoke')
            result=smoke.smoke(ssh_config=config,alias='bpi-customer-0845',components=manifest,
                components_sha256=entry['manifest']['sha256'],output=case/'smoke',duration=90,file_mib=16)
            record['smoke_all_passed']=result['ok']
            remote=result.get('remote') or {}
            record['identity_verified']=remote.get('identity_verified',False)
            record['cpu_passed']=remote.get('cpu',{}).get('ok',False)
            record['file_passed']=remote.get('file',{}).get('ok',False)
            record['failures']=remote.get('failures',[])
            record['events'][-1]['finished_unix']=time.time()
            save(OUT/'summary.json',summary)
            command('return',[PY,'-B',str(HERE/'return_to_rescue.py'),str(case/'return')],360)
            recovery=json.loads((case/'return/report.json').read_text())
            if not recovery.get('recovery_verified') or not recovery.get('sd_prefix_unchanged'):
                raise RuntimeError('回救援或 SD 前綴未核對，不進下一套')
            rescue_config=recovery['ssh_config']
            record.update(status='collected',recovery_verified=True,finished_unix=time.time(),
                          all_hardware_tests_passed=False,desktop_login_tested=False)
            save(OUT/'summary.json',summary)
            print(json.dumps({'case':key,'status':record['status'],'smoke_all_passed':record['smoke_all_passed']},ensure_ascii=False),flush=True)
        summary['status']='collected'
    except Exception as error:
        summary.update(status='interrupted',error=str(error))
        if summary['cases']:
            summary['cases'][-1].update(status='interrupted',error=str(error),finished_unix=time.time())
        raise
    finally:
        summary['finished_unix']=time.time()
        save(OUT/'summary.json',summary)


if __name__=='__main__':
    main()
