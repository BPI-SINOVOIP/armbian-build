"""0845 的有界救援冷啟動實驗；不部署客戶映像、不寫 eMMC。"""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]
sys.path.insert(0, str(ROOT / 'tools'))
from bpi_h618_rescue_boot import RecordedChannel, boot, input_manifest, BRIDGE_SHA
from bpi_sram_uart import open_serial
from bpi_h618_rescue.bpi_rescue_cli import ChineseArgumentParser
from rescue_network import connect


def power(action, expected, records):
    result = subprocess.run(['bpi-pw','--device','bpi-pw-1',action],capture_output=True,timeout=60)
    record = json.loads(result.stdout if result.returncode == 0 else result.stderr)
    records.append({'command':action,'returncode':result.returncode,'result':record})
    device=record.get('device',{})
    if not (result.returncode==0 and record.get('ok') is True and record.get('verified') is True
            and device.get('identity_verified') is True and device.get('name')=='bpi-pw-1'
            and device.get('mac')=='EC:B9:31:24:F7:D1' and device.get('ip')=='192.168.50.245'
            and device.get('on') is expected):
        raise RuntimeError('電源回讀或資產身分不符，不繼續操作')
    return record


def shutdown(console, source):
    identity=console.run_shell('uname -r',timeout=10)
    if identity.exitcode or identity.output.strip()!=b'6.6.75-current-sunxi64':
        raise RuntimeError('目前核心不是預期救援基線')
    if source=='linux':
        processes=console.run_shell('pgrep -x stress-ng',timeout=10)
        if processes.exitcode!=1:
            raise RuntimeError('原 Linux 尚有壓測或狀態不明，不關機')
        root=console.run_shell('findmnt -n -o SOURCE /',timeout=10)
        if root.exitcode or root.output.strip()!=b'/dev/mmcblk0p1':
            raise RuntimeError('原 SD 根系統不符')
        console.send('systemctl poweroff\n')
    else:
        check=console.run_shell('bpi-rescue inventory',timeout=15)
        if check.exitcode:
            raise RuntimeError('未核對獨立救援，不關機')
        inventory=json.loads(check.output)
        if '/dev/mmcblk' in inventory['mountinfo'] or len(inventory['swaps'].splitlines())>1:
            raise RuntimeError('救援尚有媒體掛載或 swap，不直接關機')
        console.send('/bin/busybox poweroff -f\n')
    console.expect_literal('reboot: Power down',timeout=180)


def main():
    parser=ChineseArgumentParser(description=__doc__)
    parser.add_argument('--from-state',choices=('linux','rescue'),required=True)
    parser.add_argument('--count',type=int,default=1)
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args()
    if not 1<=args.count<=10: raise RuntimeError('次數必須為 1 至 10')
    args.output=args.output.absolute()
    args.output.mkdir(mode=0o700,exist_ok=False)
    bridge=(ROOT/'output/evidence/bpi-sram-a1-fit-2048-002/bridge/bridge-package.bin').read_bytes()
    if hashlib.sha256(bridge).hexdigest()!=BRIDGE_SHA: raise RuntimeError('橋接封包不符')
    inputs,prefix=input_manifest(HERE/'rescue-inputs-002.json')
    summary={'ok':False,'cycles':[],'media_written':False,'sd_prefix_before':prefix}
    try:
        for number in range(1,args.count+1):
            output=args.output/f'cycle-{number:02d}'
            output.mkdir(mode=0o700)
            record={'ok':False,'started_unix':time.time(),'commands':[],'power':[]}
            summary['cycles'].append(record)
            try:
                with open_serial('/dev/ttyUSB0',115200,10) as port:
                    with RecordedChannel(port,log_path=output/'uart.bin') as console:
                        power('status',True,record['power'])
                        shutdown(console,args.from_state if number==1 else 'rescue')
                        record['shutdown_verified']=True
                        power('off',False,record['power'])
                        off=time.monotonic()
                        power('status',False,record['power'])
                        time.sleep(max(0,10-(time.monotonic()-off)))
                        record['off_seconds']=time.monotonic()-off
                        power('on',True,record['power'])
                        console.expect_regex(rb'BPI-SUP1 [^\r\n]*event=ready[^\r\n]*\r?\n',timeout=20)
                        record.update(boot(console,bridge,inputs,record['commands']))
                        record.update(connect(console,output))
                record['ok']=True
            except Exception as error:
                record['error']=str(error)
                raise
            finally:
                record['finished_unix']=time.time()
                with (output/'report.json').open('x') as stream:
                    json.dump(record,stream,ensure_ascii=False,indent=2)
                    stream.flush(); os.fsync(stream.fileno())
                print(json.dumps({'cycle':number,'ok':record['ok'],'error':record.get('error')},ensure_ascii=False),flush=True)
        summary['ok']=True
    finally:
        with (args.output/'summary.json').open('x') as stream:
            json.dump(summary,stream,ensure_ascii=False,indent=2)
            stream.flush(); os.fsync(stream.fileno())
    return 0


if __name__=='__main__':
    raise SystemExit(main())
