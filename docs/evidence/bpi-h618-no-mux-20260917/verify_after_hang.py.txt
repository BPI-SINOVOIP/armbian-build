"""沿用已審查的身分與有界回讀函式，唯讀核對異常斷電後映像。"""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time

HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[3]
sys.path.insert(0,str(ROOT/'tools'))
import bpi_h618_emmc_deploy as deploy

CASE=ROOT/'output/evidence/bpi-h618-no-mux/T4-matrix-004/noble-xfce_desktop'
OUT=CASE/'readback-after-hang-001'
OUT.mkdir(mode=0o700,exist_ok=False)
receipt=json.loads((CASE/'deploy/receipt.json').read_text())
recovery=json.loads((CASE/'return-after-hang-001/report.json').read_text())
if receipt.get('status')!='verified' or not recovery.get('recovery_verified') or not recovery.get('sd_prefix_unchanged'):
    raise RuntimeError('原部署或回救援證據不足')
source=deploy.remote_program()
code="""
import json,os,signal,time
namespace={'__name__':'_bpi_readback'}
exec(SOURCE,namespace)
signal.alarm(240)
require=namespace['require']
deadline=time.monotonic()+230
def check():
    require(time.monotonic()<deadline,'唯讀核對逾時')
rescue=namespace['rescue_identity']()
identity=namespace['checks']['inspect'](EXPECTED)
fd=os.open(identity['device'],os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
try:
    namespace['checks']['check_fd'](fd,identity)
    raw=namespace['hash_range'](fd,RAW['bytes'],check)
    require(namespace['checks']['inspect'](EXPECTED)==identity,'回讀後媒體身分改變')
    require(raw==RAW,'異常斷電後內容與原映像不同')
    print(json.dumps({'ok':True,'rescue':rescue,'identity':identity,'raw':raw,'media_written':False}))
finally:
    os.close(fd)
"""
program='SOURCE='+repr(source)+'\nEXPECTED='+repr(receipt['request']['expected'])+'\nRAW='+repr(receipt['source']['raw'])+'\n'+code
report={'ok':False,'started_unix':time.time(),'media_written':False,
    'program_sha256':hashlib.sha256(program.encode()).hexdigest(),
    'source_receipt_sha256':hashlib.sha256((CASE/'deploy/receipt.json').read_bytes()).hexdigest()}
try:
    result=subprocess.run(['ssh','-F',recovery['ssh_config'],'bpi-rescue-0845','python3 -B -'],
        input=program.encode(),capture_output=True,timeout=260)
    report.update(exitcode=result.returncode,stderr=result.stderr.decode(errors='replace'))
    if result.returncode: raise RuntimeError('遠端唯讀範圍核對失敗')
    remote=json.loads(result.stdout)
    if not remote.get('ok') or remote.get('raw')!=receipt['source']['raw']:
        raise RuntimeError('回讀摘要未通過')
    report.update(ok=True,remote=remote)
finally:
    report['finished_unix']=time.time()
    with (OUT/'report.json').open('x') as stream: json.dump(report,stream,ensure_ascii=False,indent=2)
    print(json.dumps(report,ensure_ascii=False),flush=True)
