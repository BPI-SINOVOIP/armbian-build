"""0845 救援連線：認證只由主機既有設定注入 RAM，經 UART 固定新 SSH 公鑰。"""
import hashlib
import ipaddress
import json
import os
from pathlib import Path
import re
import shlex
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT / "tools"))
from bpi_lab_console import ConsoleSession
from bpi_sram_uart import open_serial

SCRIPT = r'''
import hashlib,os,sys,termios
fd=sys.stdin.fileno()
previous=termios.tcgetattr(fd)
changed=termios.tcgetattr(fd)
changed[3]&=~termios.ECHO
try:
 termios.tcsetattr(fd,termios.TCSANOW,changed)
 print('BPI_WIFI_SECRET_READY',flush=True)
 password=sys.stdin.buffer.readline(129).rstrip(b'\r\n')
 if not 8<=len(password)<=64: raise ValueError('認證長度不符')
 if len(password)==64:
  int(password,16); psk=password.decode('ascii')
 else:
  psk=hashlib.pbkdf2_hmac('sha1',password,b'NA3',4096,32).hex()
 content='ctrl_interface=/run/wpa_supplicant\nnetwork={\n ssid="NA3"\n psk='+psk+'\n}\n'
 out=os.open('/run/wpa_supplicant.conf',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
 with os.fdopen(out,'w') as stream: stream.write(content)
finally:
 termios.tcsetattr(fd,termios.TCSANOW,previous)
print('BPI_WIFI_SECRET_STORED',flush=True)
'''


def connect(console, output):
    identity = console.run_shell("cat /etc/bpi-rescue.json", timeout=10)
    if identity.exitcode or json.loads(identity.output) != {"schema":"bpi-h618-rescue-v1","kernel":"6.6.75-current-sunxi64"}:
        raise RuntimeError("未進入指定救援系統")
    response = subprocess.run(["sudo","-n","nmcli","--show-secrets","-g",
                               "802-11-wireless-security.psk","connection","show","NA3"],
                              capture_output=True,check=True,timeout=10)
    secret = response.stdout.strip()
    if not 8<=len(secret)<=64 or b"\n" in secret or b"\r" in secret:
        raise RuntimeError("主機認證格式不符")
    console.send("python3 -B -c "+shlex.quote("exec("+repr(SCRIPT)+")")+"\n")
    console.expect_regex(rb"(?:^|\r?\n)BPI_WIFI_SECRET_READY\r?\n",timeout=10)
    console.send(secret+b"\n",secret=True)
    del secret,response
    console.expect_regex(rb"(?:^|\r?\n)BPI_WIFI_SECRET_STORED\r?\n",timeout=10)
    network=console.run_shell("bpi-rescue wifi --interface wlan0 --config /run/wpa_supplicant.conf",timeout=100)
    if network.exitcode: raise RuntimeError("救援 Wi-Fi 連線未成功")
    return bind_ssh(console, output)


def bind_ssh(console, output):
    result=console.run_shell("/usr/bin/ip -j address show dev wlan0",timeout=10)
    if result.exitcode: raise RuntimeError("救援位址查詢失敗")
    addresses=[v['local'] for device in json.loads(result.output) for v in device['addr_info']
               if v['family']=='inet' and v.get('scope')=='global']
    if len(addresses)!=1 or ipaddress.ip_address(addresses[0]) not in ipaddress.ip_network('192.168.50.0/24'):
        raise RuntimeError("救援位址不在預期網路")
    key=console.run_shell("cat /run/ssh/ssh_host_ed25519_key.pub",timeout=15)
    fields=key.output.strip().split()
    if key.exitcode or len(fields)<2 or fields[0]!=b'ssh-ed25519' or not re.fullmatch(rb'[A-Za-z0-9+/=]+',fields[1]):
        raise RuntimeError("未取得救援 SSH 主機公鑰")
    keytext=b' '.join(fields[:2]).decode()
    known=output/'known_hosts'
    with known.open('x') as stream: stream.write(addresses[0]+' '+keytext+'\n')
    known.chmod(0o600)
    config=output/'ssh-config'
    values={"HostName":addresses[0],"User":"root","IdentityFile":str(Path(__file__).parent/'ssh/id_ed25519'),
            "UserKnownHostsFile":str(known),"GlobalKnownHostsFile":"/dev/null","StrictHostKeyChecking":"yes",
            "IdentitiesOnly":"yes","BatchMode":"yes","HostKeyAlgorithms":"ssh-ed25519",
            "ClearAllForwardings":"yes","ForwardAgent":"no","RequestTTY":"no","ControlMaster":"no",
            "ControlPath":"none","ConnectTimeout":"10","ServerAliveInterval":"15","ServerAliveCountMax":"3"}
    with config.open('x') as stream:
        stream.write('Host bpi-rescue-0845\n'+''.join('    '+key+' '+value+'\n' for key,value in values.items()))
    config.chmod(0o600)
    response=subprocess.run(['ssh','-F',str(config),'bpi-rescue-0845','bpi-rescue inventory'],
                            capture_output=True,timeout=30)
    with (output/'ssh-inventory.json').open('xb') as stream: stream.write(response.stdout)
    if response.returncode: raise RuntimeError("嚴格 SSH 驗證失敗")
    return {"network_verified":True,"strict_ssh_verified":True,"address":addresses[0],
            "hostkey":keytext,"ssh_config":str(config),"inventory_sha256":hashlib.sha256(response.stdout).hexdigest()}


if __name__ == '__main__':
    output=Path(sys.argv[1]).absolute()
    output.mkdir(mode=0o700,exist_ok=False)
    record={"ok":False}
    try:
        with open_serial('/dev/ttyUSB0',115200,10) as port:
            with ConsoleSession(port,log_path=output/'uart.bin') as console:
                record.update(connect(console,output))
        record['ok']=True
    finally:
        with (output/'report.json').open('x') as stream: json.dump(record,stream,ensure_ascii=False,indent=2)
        print(json.dumps(record,ensure_ascii=False))
