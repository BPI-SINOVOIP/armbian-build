"""從獨立救援冷啟動已核對部署的客戶系統，保留同一 UART 紀錄。"""
import hashlib
import json
import os
from pathlib import Path
import secrets
import sys
import time

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]
sys.path.insert(0, str(ROOT/'tools'))
from bpi_h618_customer_boot import boot, load_inputs
from bpi_h618_rescue_boot import RecordedChannel, BRIDGE_SHA
from bpi_sram_uart import open_serial
from bpi_h618_rescue.bpi_rescue_cli import ChineseArgumentParser
from rescue_cycles import power, shutdown


def initialize(console, output):
    password=secrets.token_urlsafe(24).encode()
    fd=os.open(output/'root-password',os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
    with os.fdopen(fd,'wb') as stream: stream.write(password+b'\n')
    console.expect_literal('Create root password:',timeout=40)
    console.send(password+b'\n',secret=True)
    console.expect_literal('Repeat root password:',timeout=15)
    console.send(password+b'\n',secret=True)
    prompt=console.expect_regex(rb'(?:2\) [^\r\n]+\r?\n|Please provide a username[^\r\n]*: ?)',timeout=40)
    if prompt.matched.startswith(b'2)'):
        if b'1) bash' not in prompt.before: raise RuntimeError('未確認 bash 選項')
        console.send('1\n')
        console.expect_literal('Please provide a username',timeout=15)
    console.send(b'\x03')
    console.expect_regex(rb'root@bananapim4zeroemac:[^\r\n]*# ?',timeout=20)
    result=console.run_shell('id -u',timeout=10)
    if result.exitcode or result.output.strip()!=b'0': raise RuntimeError('未取得 root shell')
    return {'root_initialized':True,'ordinary_user_created':False,'desktop_login_tested':False}


def main():
    parser = ChineseArgumentParser(description=__doc__)
    parser.add_argument('--components', type=Path, required=True)
    parser.add_argument('--components-sha256', required=True)
    parser.add_argument('--receipt', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--initialize', action='store_true')
    args = parser.parse_args()
    components, receipt, evidence = load_inputs(args.components, args.components_sha256, args.receipt)
    bridge = (ROOT/'output/evidence/bpi-sram-a1-fit-2048-002/bridge/bridge-package.bin').read_bytes()
    if hashlib.sha256(bridge).hexdigest() != BRIDGE_SHA:
        raise RuntimeError('橋接封包不符')
    args.output.mkdir(mode=0o700, exist_ok=False)
    report = {'ok':False, 'commands':[], 'power':[], 'evidence':evidence,
              'components':components, 'started_unix':time.time(),
              'original_boot_chain_verified':False, 'sd_function_tested':False,
              'ram_dtb_changes':{'/soc/mmc@4020000/status':'disabled'},
              'root_cid_verified':False, 'system_verified':False}
    try:
        with open_serial('/dev/ttyUSB0', 115200, 10) as port:
            with RecordedChannel(port, log_path=args.output/'uart.bin') as console:
                power('status', True, report['power'])
                shutdown(console, 'rescue')
                report['shutdown_verified'] = True
                power('off', False, report['power'])
                off = time.monotonic()
                power('status', False, report['power'])
                time.sleep(max(0, 10-(time.monotonic()-off)))
                report['off_seconds'] = time.monotonic()-off
                power('on', True, report['power'])
                console.expect_regex(rb'BPI-SUP1 [^\r\n]*event=ready[^\r\n]*\r?\n', timeout=20)
                report.update(boot(console, bridge, components, receipt, report['commands']))
                if args.initialize:
                    report.update(initialize(console,args.output))
        report['ok'] = True
    except Exception as error:
        report['error'] = str(error)
        raise
    finally:
        report['finished_unix'] = time.time()
        with (args.output/'report.json').open('x') as stream:
            json.dump(report, stream, ensure_ascii=False, indent=2)
            stream.flush()
            os.fsync(stream.fileno())
        print(json.dumps({k:report.get(k) for k in ('ok','error','status','system_verified')}, ensure_ascii=False))


if __name__ == '__main__':
    main()
